@tool
extends Node3D
class_name IslandSpawner

const ISLAND_GENERATOR = preload("uid://blnohhm7hi5pt")

@export_group("Map Controls")
## Click this to generate a completely new map with a random seed
@export var generate_map: bool = false:
	set(value):
		if value:
			# Randomize the seed every time we click generate
			world_seed = randi() 
			_generate_grid()
			generate_map = false

@export_group("Grid Settings")
@export var world_seed: int = 42
@export var grid_width: int = 5
@export var grid_depth: int = 5

## The maximum physical footprint of an island inside the cell
@export var cell_size: float = 200.0
## The guaranteed empty ocean space between grid cells
@export var cell_margin: float = 50.0

@export_group("Spawn Rules")
## How full the ocean is (0.0 = completely empty, 1.0 = an island in every cell)
@export_range(0.0, 1.0) var island_density: float = 0.4

## The chances of an island being a certain size (These normalize automatically)
@export_range(0.0, 1.0) var small_island_chance: float = 0.5
@export_range(0.0, 1.0) var medium_island_chance: float = 0.3
@export_range(0.0, 1.0) var large_island_chance: float = 0.2

func _ready() -> void:
	await get_tree().process_frame
	_generate_grid()

func _generate_grid() -> void:
	# 1. Cleanup old map
	for child in get_children():
		child.queue_free()

	var rng = RandomNumberGenerator.new()
	rng.seed = world_seed

	# 2. Combine size and margin to get the total stepping distance between centers
	var total_spacing = cell_size + cell_margin

	# Calculate the exact center offset for positioning using the new total_spacing
	var offset_x = (grid_width - 1) / 2.0 * total_spacing
	var offset_z = (grid_depth - 1) / 2.0 * total_spacing
	
	var center_x = grid_width / 2
	var center_z = grid_depth / 2

	# 3. Loop through grid
	for x in range(grid_width):
		for z in range(grid_depth):
			
			var is_center_tile = (x == center_x and z == center_z)
			
			# 4. Density Check
			if not is_center_tile and rng.randf() > island_density:
				continue 
			
			# 5. Determine island type based on the size dice roll
			var size_roll = rng.randf()
			var total_chance = small_island_chance + medium_island_chance + large_island_chance
			var normalized_roll = size_roll * total_chance 
			
			var role_name = ""
			var color = Color()
			
			if normalized_roll < small_island_chance:
				role_name = "Small_Island"
				color = Color.BLUE
			elif normalized_roll < small_island_chance + medium_island_chance:
				role_name = "Medium_Island"
				color = Color.GREEN
			else:
				role_name = "Large_Island"
				color = Color.ORANGE
				
			# 6. Apply the offset using our total_spacing math
			var spawn_position = Vector3(
				(x * total_spacing) - offset_x, 
				0, 
				(z * total_spacing) - offset_z
			)
			
			_spawn_island(spawn_position, role_name, rng.randi())
			#_spawn_placeholder(spawn_position, role_name,color)

func _spawn_island(pos: Vector3, role: String, unique_seed: int) -> void:
	# 1. Instantiate the actual IslandGenerator scene
	var island_instance = ISLAND_GENERATOR.instantiate()
	island_instance.name = role
	
	# 2. Calculate physical size based purely on cell_size
	var current_size: float = 0.0
	if role == "Small_Island":
		current_size = cell_size * 0.4 # 40% of the cell
	elif role == "Medium_Island":
		current_size = cell_size * 0.7 # 70% of the cell
	elif role == "Large_Island":
		current_size = cell_size * 1.0 # 100% of the cell footprint
		
	# 3. Pass the variables to your IslandGenerator script
	island_instance.island_size = current_size
	island_instance.island_seed = unique_seed
	
	# 4. Add it to the tree and set its position
	add_child(island_instance)
	island_instance.position = pos
	
	# 5. Trigger the generation logic
	island_instance.generate = true
	
	# 6. Save the instanced scene to the editor tree
	if Engine.is_editor_hint() and get_tree() and get_tree().edited_scene_root:
		island_instance.owner = get_tree().edited_scene_root
